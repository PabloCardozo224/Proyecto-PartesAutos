

create procedure sp_clientes_insertar(
    in p_nombre varchar(35),
    in p_apellido varchar(35),
    in p_dni int,
    in p_telefono int,
    in p_direccion varchar(25)
)
begin
    insert into clientes (nombre, apellido, dni, telefono, direccion)
    values (p_nombre, p_apellido, p_dni, p_telefono, p_direccion);
end //

delimiter ;


delimiter //

create procedure sp_clientes_actualizar_contacto(
    in p_id_cliente int,
    in p_telefono int,
    in p_direccion varchar(25)
)
begin
    update clientes 
    set telefono = p_telefono, direccion = p_direccion 
    where id_cliente = p_id_cliente;
end //

delimiter ;


delimiter //

create procedure sp_clientes_buscar_por_dni(
    in p_dni int
)
begin
    select * from clientes where dni = p_dni;
end //

delimiter ;


delimiter //

create procedure sp_clientes_total_gastado(
    in p_id_cliente int,
    out p_total_gastado float
)
begin
    select coalesce(sum(pg.monto), 0) into p_total_gastado
    from envios e
    inner join ventas v on e.id_envio = v.id_envio
    inner join pagos pg on v.id_pago = pg.id_pago
    where e.id_cliente = p_id_cliente and pg.estado = 'confirmado';
end //

delimiter ;


delimiter //

create procedure sp_clientes_eliminar(
    in p_id_cliente int
)
begin
    declare v_cant_envios int;
    
    select count(*) into v_cant_envios 
    from envios 
    where id_cliente = p_id_cliente;
    
    if v_cant_envios > 0 then
        select 'No se puede eliminar el cliente porque tiene envíos registrados.' as mensaje;
    else
        delete from clientes where id_cliente = p_id_cliente;
        select 'Cliente eliminado correctamente.' as mensaje;
    end if;
end //

delimiter ;


delimiter //

create procedure sp_clientes_buscar_por_id(
    in p_id_cliente int
)

    select * from clientes where id_cliente = p_id_cliente;
end //

delimiter ;


delimiter //

create procedure sp_clientes_listar_todos()

end //

delimiter ;


delimiter //

create procedure sp_clientes_listar_ordenados()
begin
    select * from clientes order by apellido asc, nombre asc;
end //

delimiter ;


delimiter //

create procedure sp_clientes_buscar_por_nombre(
    in p_criterio varchar(35)
)
begin
    select * from clientes 
    where nombre like concat('%', p_criterio, '%') 
       or apellido like concat('%', p_criterio, '%');
end //

delimiter ;


delimiter //

create procedure sp_clientes_buscar_por_direccion(
    in p_calle varchar(25)
)
begin
    select * from clientes where direccion like concat('%', p_calle, '%');
end //

delimiter ;


delimiter //

create procedure sp_clientes_buscar_por_telefono(
    in p_telefono int
)
begin
    select * from clientes where telefono = p_telefono;
end //

delimiter ;
