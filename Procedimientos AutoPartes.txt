CREATE PROCEDURE sp_Marcas_ListarConProductos()
BEGIN
    SELECT m.id_marca, m.nombre, COUNT(p.Id_Producto) AS total_productos
    FROM Marcas m
    LEFT JOIN Productos p ON m.id_marca = p.id_marca
    GROUP BY m.id_marca, m.nombre;
END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE sp_Marcas_ListarSinProductos()
BEGIN
    SELECT m.*
    FROM Marcas m
    LEFT JOIN Productos p ON m.id_marca = p.id_marca
    WHERE p.Id_Producto IS NULL;
END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE sp_Marcas_InsertarSiNoExiste(IN p_nombre VARCHAR(50))
BEGIN
    INSERT INTO Marcas(nombre)
    SELECT p_nombre FROM DUAL
    WHERE NOT EXISTS (SELECT 1 FROM Marcas WHERE nombre = p_nombre);
END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE sp_Marcas_ListarTopPorVentas()
BEGIN
    SELECT m.id_marca, m.nombre,
           COALESCE(SUM(v.cantidad_productos), 0) AS unidades_vendidas,
           COALESCE(SUM(v.cantidad_productos * p.precio), 0) AS facturado
    FROM Marcas m
    LEFT JOIN Productos p ON m.id_marca = p.id_marca
    LEFT JOIN Ventas v    ON v.Id_Producto = p.Id_Producto
    GROUP BY m.id_marca, m.nombre
    ORDER BY unidades_vendidas DESC;
END //
DELIMITER ;

CREATE PROCEDURE sp_Empleados_TopVentas(IN p_limite INT)
BEGIN
    SELECT e.Id_Empleado, e.Nombre, e.Apellido, e.Puesto,
           COUNT(v.id_Ventas) AS total_ventas,
           COALESCE(SUM(v.cantidad_productos), 0) AS unidades_vendidas
    FROM Empleados e
    LEFT JOIN Ventas v ON e.Id_Empleado = v.id_Empleado
    GROUP BY e.Id_Empleado, e.Nombre, e.Apellido, e.Puesto
    ORDER BY total_ventas DESC
    LIMIT p_limite;
END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE sp_Empleados_VentasPorEmpleado(IN p_id INT)
BEGIN
    SELECT v.id_Ventas, v.Fecha_venta, v.Tipo_pedido,
           v.cantidad_productos, p.Nombre AS producto, p.precio
    FROM Ventas v
    INNER JOIN Productos p ON v.Id_Producto = p.Id_Producto
    WHERE v.id_Empleado = p_id
    ORDER BY v.Fecha_venta DESC;
END //
DELIMITER ;

CREATE PROCEDURE sp_Ventas_Insertar(IN p_Fecha_venta DATETIME, IN p_cantidad_productos INT, IN p_Tipo_pedido VARCHAR(30), IN p_Id_Producto INT, IN p_id_Empleado INT, IN p_ID_Envio INT, IN p_Id_Pago INT)

CREATE PROCEDURE sp_Ventas_DetalleCompleto(IN p_id INT)
BEGIN
    SELECT v.id_Ventas, v.Fecha_venta, v.Tipo_pedido, v.cantidad_productos,
           p.Nombre AS producto, p.Modelo, p.precio,
           (p.precio * v.cantidad_productos) AS subtotal,
           CONCAT(e.Nombre, ' ', e.Apellido) AS empleado,
           pg.Metodo AS metodo_pago, pg.Estado AS estado_pago,
           en.nombre_calle, en.nombre_localidad, en.nombre_provincia,
           CONCAT(c.Nombre, ' ', c.Apellido) AS cliente
    FROM Ventas v
    INNER JOIN Productos  p  ON v.Id_Producto = p.Id_Producto
    INNER JOIN Empleados  e  ON v.id_Empleado = e.Id_Empleado
    INNER JOIN Pagos      pg ON v.Id_Pago      = pg.Id_Pago
    INNER JOIN Envios     en ON v.ID_Envio     = en.ID_Envio
    LEFT  JOIN Clientes   c  ON c.id_Ventas    = v.id_Ventas
    WHERE v.id_Ventas = p_id;
END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE sp_Ventas_TotalFacturado()
BEGIN
    SELECT COALESCE(SUM(p.precio * v.cantidad_productos), 0) AS total_facturado
    FROM Ventas v
    INNER JOIN Productos p ON v.Id_Producto = p.Id_Producto;
END //
DELIMITER ;

CREATE PROCEDURE sp_Clientes_TopCompras(IN p_limite INT)
BEGIN
    SELECT c.Id_cliente, c.Nombre, c.Apellido, c.DNI,
           COUNT(v.id_Ventas) AS total_compras,
           COALESCE(SUM(p.precio * v.cantidad_productos), 0) AS total_gastado
    FROM Clientes c
    INNER JOIN Ventas    v ON c.id_Ventas   = v.id_Ventas
    INNER JOIN Productos p ON v.Id_Producto = p.Id_Producto
    GROUP BY c.Id_cliente, c.Nombre, c.Apellido, c.DNI
    ORDER BY total_gastado DESC
    LIMIT p_limite;
END //
DELIMITER ;
