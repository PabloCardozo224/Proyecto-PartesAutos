create procedure sp_clientes_total_gastado(
    in p_id_cliente int,
    out p_total_gastado float
)
begin
    select coalesce(sum(pg.monto), 0) into p_total_gastado
    from envios e
    inner join ventas v on e.id_envio = v.id_envio
    inner join pagos pg on v.id_pago = pg.id_pago
    where e.id_cliente = p_id_cliente and pg.estado = 'confirmado';
end //

delimiter ;


delimiter //

create procedure sp_clientes_eliminar(
    in p_id_cliente int
)
begin
    declare v_cant_envios int;
    
    select count(*) into v_cant_envios 
    from envios 
    where id_cliente = p_id_cliente;
    
    if v_cant_envios > 0 then
        select 'No se puede eliminar el cliente porque tiene envíos registrados.' as mensaje;
    else
        delete from clientes where id_cliente = p_id_cliente;
        select 'Cliente eliminado correctamente.' as mensaje;
    end if;
end //

delimiter ;


delimiter //

create procedure sp_clientes_listar_compras(
    in p_id_cliente int
)
begin
    select v.id_venta, v.f_venta, p.nombre as producto, v.cant_prod, pg.monto
    from envios e
    inner join ventas v on e.id_envio = v.id_envio
    inner join productos p on v.id_producto = p.id_producto
    inner join pagos pg on v.id_pago = pg.id_pago
    where e.id_cliente = p_id_cliente;
end //

delimiter ;


delimiter //

create procedure sp_clientes_con_envios_activos()
begin
    select distinct c.*
    from clientes c
    inner join envios e on c.id_cliente = e.id_cliente
    where e.estado_env in ('en camino', 'en almacen', 'reenviando');
end //

delimiter ;


delimiter //

create procedure sp_clientes_sin_compras()
begin
    select c.* 
    from clientes c
    left join envios e on c.id_cliente = e.id_cliente
    where e.id_envio is null;
end //

delimiter ;


delimiter //

create procedure sp_clientes_cantidad_pedidos(
    in p_id_cliente int,
    out p_cantidad int
)
begin
    select count(distinct id_pedido) into p_cantidad
    from envios
    where id_cliente = p_id_cliente;
end //

delimiter ;


delimiter //

create procedure sp_clientes_mas_comprador()
begin
    select c.id_cliente, c.nombre, c.apellido, count(v.id_venta) as total_compras
    from clientes c
    inner join envios e on c.id_cliente = e.id_cliente
    inner join ventas v on e.id_envio = v.id_envio
    group by c.id_cliente, c.nombre, c.apellido
    order by total_compras desc
    limit 1;
end //

delimiter ;


delimiter //

create procedure sp_productos_insertar(
    in p_nombre varchar(35),
    in p_descripcion varchar(50),
    in p_precio int,
    in p_cantidad int,
    in p_tamaño enum("chico", "mediano", "grande"),
    in p_modelo varchar(40),
    in p_color varchar(25),
    in p_tipo_parte varchar(100)
)
begin
    insert into productos (nombre, descripcion, precio, cantidad, tamaño, modelo, color, tipo_parte)
    values (p_nombre, p_descripcion, p_precio, p_cantidad, p_tamaño, p_modelo, p_color, p_tipo_parte);

delimiter ;


delimiter //

create procedure sp_productos_actualizar_stock(
    in p_id_producto int,
    in p_cambio_cantidad int,
    out p_stock_restante int
)
begin
    update productos 
    set cantidad = cantidad + p_cambio_cantidad 
    where id_producto = p_id_producto;

    select cantidad into p_stock_restante 
    from productos 
    where id_producto = p_id_producto;
end //

delimiter ;


delimiter //

create procedure sp_productos_aumentar_precio_por_tipo(
    in p_tipo_parte varchar(100),
    in p_porcentaje float
)
begin
    update productos 
    set precio = round(precio * (1 + (p_porcentaje / 100)))
    where tipo_parte = p_tipo_parte;
end //

delimiter ;


delimiter //
